# Internationalization (i18n)

Multi-language support for img2prompt extension.

## Supported Languages

| Language Code | Language | Status |
|--------------|----------|--------|
| `en` | English | ✅ Primary |
| `zh` | Chinese (Simplified) | ✅ Primary |
| `zh-TW` | Chinese (Traditional) | 📋 Planned |
| `ja` | Japanese | 📋 Planned |
| `ko` | Korean | 📋 Planned |

## WXT i18n Support

WXT provides built-in i18n support through the WebExtension API.

```
public/
```

## Usage in Code

### Helper Function

```typescript
// lib/i18n.ts
export function t(key: string, substitutions?: string[]): string {
  return browser.i18n.getMessage(key, substitutions);
}

// Usage
const buttonText = t('selectImage'); // Returns localized string
```

### React Component

```typescript
// components/SelectionButton.tsx
import { t } from '@/lib/i18n';

function SelectionButton() {
  return (
    <button onClick={handleSelect}>
      {t('selectImage')}
    </button>
  );
}
```

## Detect Browser Language

```typescript
// Get browser UI language
const browserLanguage = browser.i18n.getUILanguage(); // 'en', 'zh-CN', etc.

// Map to supported languages
function mapToSupportedLanguage(lang: string): SupportedLanguage {
  const mapping: Record<string, SupportedLanguage> = {
    'en': 'en',
    'en-US': 'en',
    'en-GB': 'en',
    'zh': 'zh',
    'zh-CN': 'zh',
    'zh-Hans': 'zh',
    'zh-TW': 'zh-TW',
    'zh-Hant': 'zh-TW',
    'ja': 'ja',
    'ko': 'ko',
  };

  return mapping[lang] || 'en';
}

const userLanguage = mapToSupportedLanguage(browserLanguage);
```

## Language Switching

### Store Preference

```typescript
// lib/storage.ts
interface ExtensionConfig {
  // ... other config
  general: {
    language: 'en' | 'zh' | 'ja' | 'ko' | 'zh-TW';
    autoDetect: boolean;
  };
}

// Save language preference
async function setLanguage(language: SupportedLanguage) {
  await browser.storage.local.set({
    general: {
      language,
      autoDetect: false,
    },
  });

  // Reload UI
  location.reload();
}
```

### Settings UI

```typescript
// options/App.tsx
function LanguageSelector() {
  const [language, setLanguage] = useState('en');

  const handleChange = async (newLang: string) => {
    await setLanguage(newLang);
    setLanguage(newLang);
  };

  return (
    <div>
      <label>Language / 语言</label>
      <select value={language} onChange={(e) => handleChange(e.target.value)}>
        <option value="en">English</option>
        <option value="zh">中文 (简体)</option>
        <option value="zh-TW">中文 (繁體)</option>
        <option value="ja">日本語</option>
        <option value="ko">한국어</option>
      </select>
    </div>
  );
}
```

## Best Practices

### 1. Always Use i18n

```typescript
// ✅ Good
const message = t('errorNetwork');

// ❌ Bad
const message = 'Network error';
```

### 2. Fallback for Missing Keys

```typescript
function t(key: string): string {
  const message = browser.i18n.getMessage(key);
  return message || key; // Fallback to key name
}
```

### 3. Don't Concatenate Strings

```typescript
// ❌ Bad - breaks i18n
const message = t('error') + ': ' + t('networkIssue');

// ✅ Good - use placeholders
const message = t('errorWithDetails', [t('networkIssue')]);
```

### 4. Test All Languages

```typescript
// Test that all keys exist in all locales
const requiredKeys = [
  'extensionName',
  'selectImage',
  'screenshotMode',
  // ...
];

for (const lang of ['en', 'zh', 'zh-TW']) {
  for (const key of requiredKeys) {
    const message = browser.i18n.getMessage(key);
    if (!message) {
      console.error(`Missing key "${key}" in ${lang}`);
    }
  }
}
```
